cmake_minimum_required(VERSION 3.20)
project(ip-reporter VERSION 1.1.2 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
include(CTest)
include(GNUInstallDirs)
find_package(Threads REQUIRED)

add_library(ip_reporter_core src/config.cpp src/network.cpp src/http.cpp src/agent.cpp src/console.cpp)
target_include_directories(ip_reporter_core PUBLIC include)
target_include_directories(ip_reporter_core SYSTEM PUBLIC third_party)
target_compile_definitions(ip_reporter_core PUBLIC IP_REPORTER_VERSION="${PROJECT_VERSION}")
target_link_libraries(ip_reporter_core PUBLIC Threads::Threads)
if(WIN32)
  target_compile_definitions(ip_reporter_core PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0602)
  target_link_libraries(ip_reporter_core PUBLIC ws2_32 iphlpapi winhttp advapi32)
else()
  find_package(CURL 7.34 REQUIRED)
  target_link_libraries(ip_reporter_core PUBLIC CURL::libcurl)
endif()
if(MSVC)
  target_compile_options(ip_reporter_core PRIVATE /W4 /utf-8 /permissive-)
else()
  target_compile_options(ip_reporter_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_executable(ip-reporter src/main.cpp src/service.cpp)
target_link_libraries(ip-reporter PRIVATE ip_reporter_core)
if(MSVC)
  target_compile_options(ip-reporter PRIVATE /W4 /utf-8 /permissive-)
endif()
if(BUILD_TESTING)
  add_executable(core_tests tests/core_tests.cpp)
  target_link_libraries(core_tests PRIVATE ip_reporter_core)
  if(MSVC)
    target_compile_options(core_tests PRIVATE /utf-8)
  endif()
  add_test(NAME core COMMAND core_tests)
  add_executable(stdout_probe tests/stdout_probe.cpp)
  target_link_libraries(stdout_probe PRIVATE ip_reporter_core)
  if(MSVC)
    target_compile_options(stdout_probe PRIVATE /utf-8)
  endif()
  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_Interpreter_FOUND)
    add_test(NAME console_output COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/console_output_test.py" "$<TARGET_FILE:stdout_probe>")
    set_tests_properties(console_output PROPERTIES TIMEOUT 20)
    add_test(NAME integration COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/integration_test.py" "$<TARGET_FILE:ip-reporter>")
    set_tests_properties(integration PROPERTIES TIMEOUT 90)
    add_test(NAME receiver COMMAND "${Python3_EXECUTABLE}" -m unittest discover -s "${CMAKE_CURRENT_SOURCE_DIR}/tools" -p test_receiver.py)
    add_test(NAME heartbeat COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/heartbeat_test.py" "$<TARGET_FILE:ip-reporter>")
    set_tests_properties(heartbeat PROPERTIES TIMEOUT 60)
    if(UNIX)
      add_test(NAME installers COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/installer_lifecycle_test.py")
      set_tests_properties(installers PROPERTIES TIMEOUT 60)
    elseif(WIN32)
      add_test(NAME installers COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/windows_installer_identity_test.py" "$<TARGET_FILE:ip-reporter>")
      set_tests_properties(installers PROPERTIES TIMEOUT 60)
    endif()
  endif()
endif()

install(TARGETS ip-reporter RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY scripts/ DESTINATION share/ip-reporter/scripts USE_SOURCE_PERMISSIONS PATTERN "init-server.*" EXCLUDE)
install(DIRECTORY packaging/services/ DESTINATION share/ip-reporter/services)
install(DIRECTORY docs DESTINATION share/doc/ip-reporter)
install(FILES packaging/README.md DESTINATION share/doc/ip-reporter/packaging)
install(DIRECTORY tools/ DESTINATION share/ip-reporter/tools FILES_MATCHING PATTERN "*.py" PATTERN "__pycache__" EXCLUDE)
install(FILES examples/config.example.json DESTINATION share/doc/ip-reporter/examples)
install(FILES README.md LICENSE THIRD_PARTY_NOTICES.md DESTINATION share/doc/ip-reporter)
install(FILES third_party/nlohmann/LICENSE.MIT DESTINATION share/doc/ip-reporter/third_party/nlohmann)
install(FILES examples/config.example.json DESTINATION share/ip-reporter)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/packaging/CPackConfig.cmake")
  include(packaging/CPackConfig.cmake)
else()
  set(CPACK_PACKAGE_VERSION "${PROJECT_VERSION}")
  set(CPACK_PACKAGE_NAME ip-reporter)
  include(CPack)
endif()
